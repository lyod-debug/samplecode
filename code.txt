CREATE OR ALTER PROCEDURE audit.usp_AttrRecon_V2_SS
    @Execution_id BIGINT
AS
BEGIN
    SET NOCOUNT ON;

    ;WITH POLICY_HIGHEST_ID AS ( /* as before */ ),
    SS_ALL_RAW AS ( /* exact same 10 blocks as before — no transformation, no dedup except POLICY_HIGHEST_ID */ ),
    V2 AS ( SELECT * FROM audit.V2_ClaimAttr_Landing WHERE Execution_id = @Execution_id ),
    COMPARED AS
    (
        SELECT
            COALESCE(V.Recon_id, S.Recon_id) AS Recon_id,
            COALESCE(V.Claim_Number, S.Claim_Number) AS Claim_Number,
            COALESCE(V.LOBCode, S.LOBCode) AS LOBCode,
            COALESCE(V.Parameter_Name, S.Parameter_Name) AS Parameter_Name,
            V.V2_value, S.SS_Value,
            CASE
                WHEN V.Claim_Number IS NULL OR S.Claim_Number IS NULL THEN 'FAIL'
                WHEN UPPER(LTRIM(RTRIM(V.V2_value))) = UPPER(LTRIM(RTRIM(S.SS_Value))) THEN 'PASS'
                ELSE 'FAIL'
            END AS V2_SS_result
        FROM V2 V
        FULL OUTER JOIN SS_ALL_RAW S
            ON S.Claim_Number = V.Claim_Number AND S.Parameter_Name = V.Parameter_Name AND S.LOBCode = V.LOBCode
    )

    INSERT INTO audit.NonFinReconciliationresult
        (Execution_id, Recon_id, Parameter_Name, LOBCode, V2_SS_Count, V2_SS_Match, V2_SS_Mismatch, V2_SS_result, Created_at)
    SELECT @Execution_id, Recon_id, Parameter_Name, LOBCode,
        COUNT(1),
        SUM(CASE WHEN V2_SS_result = 'PASS' THEN 1 ELSE 0 END),
        SUM(CASE WHEN V2_SS_result = 'FAIL' THEN 1 ELSE 0 END),
        CASE WHEN SUM(CASE WHEN V2_SS_result = 'FAIL' THEN 1 ELSE 0 END) = 0 THEN 'PASS' ELSE 'FAIL' END,
        GETDATE()
    FROM COMPARED
    GROUP BY Recon_id, Parameter_Name, LOBCode;

    MERGE audit.ClaimNonFinReconciliationData AS tgt
    USING (SELECT * FROM COMPARED WHERE V2_SS_result = 'FAIL') AS src
    ON tgt.Execution_id = @Execution_id AND tgt.Recon_id = src.Recon_id
       AND tgt.Claim_Number = src.Claim_Number AND tgt.LOBCode = src.LOBCode
    WHEN MATCHED THEN
        UPDATE SET tgt.V2_value = src.V2_value, tgt.SS_value = src.SS_Value, tgt.V2_SS_result = src.V2_SS_result
    WHEN NOT MATCHED THEN
        INSERT (Execution_id, Recon_id, Claim_Number, LOBCode, Parameter_Name, V2_value, SS_value, V2_SS_result, Created_at)
        VALUES (@Execution_id, src.Recon_id, src.Claim_Number, src.LOBCode, src.Parameter_Name, src.V2_value, src.SS_Value, src.V2_SS_result, GETDATE());
END
