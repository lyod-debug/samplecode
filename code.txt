/* CloseDate */
CASE WHEN UPPER(LTRIM(RTRIM(C.CLAIM_STATUS))) = 'F'
THEN
    CASE 
        WHEN NULLIF(LTRIM(RTRIM(S.RECORD_DATE)), '') IS NULL 
         AND NULLIF(LTRIM(RTRIM(S.RECORD_TIME)), '') IS NULL 
        THEN NULL

        WHEN NULLIF(LTRIM(RTRIM(S.RECORD_DATE)), '') IS NOT NULL 
         AND NULLIF(LTRIM(RTRIM(S.RECORD_TIME)), '') IS NULL 
        THEN VARCHAR_FORMAT(DATE(S.RECORD_DATE), 'YYYY-MM-DD')

        WHEN NULLIF(LTRIM(RTRIM(S.RECORD_DATE)), '') IS NULL 
         AND NULLIF(LTRIM(RTRIM(S.RECORD_TIME)), '') IS NOT NULL 
        THEN VARCHAR_FORMAT(TIME(S.RECORD_TIME), 'HH24:MI:SS')

        ELSE VARCHAR_FORMAT(
                TIMESTAMP(DATE(S.RECORD_DATE), TIME(S.RECORD_TIME)), 
             'YYYY-MM-DD HH24:MI:SS') /* yyyy-mm-dd hh:mi:ss */

VARCHAR_FORMAT(TIMESTAMP(DATE(S.RECORD_DATE), TIME(S.RECORD_TIME)), 'YYYY-MM-DD HH24:MI:SS')
-- Result: '2026-08-22 12:30:45'
    END

CASE WHEN UPPER(LTRIM(RTRIM(C.CLAIM_STATUS))) = 'F'

    CASE 
        WHEN NULLIF(LTRIM(RTRIM(S.RECORD_DATE)), '') IS NULL

        THEN NULL

        WHEN NULLIF(LTRIM(RTRIM(S.RECORD_DATE)), '') IS NOT NULL 
         AND NULLIF(LTRIM(RTRIM(S.RECORD_TIME)), '') IS NULL 
        THEN CONVERT(VARCHAR(10), TRY_CONVERT(DATE, S.RECORD_DATE), 120)

        WHEN NULLIF(LTRIM(RTRIM(S.RECORD_DATE)), '') IS NULL 
         AND NULLIF(LTRIM(RTRIM(S.RECORD_TIME)), '') IS NOT NULL

        ELSE CONVERT(VARCHAR(19),

                + CAST(TRY_CONVERT(TIME, S.RECORD_TIME) AS DATETIME), 
             120) /* 120 = yyyy-mm-dd hh:mi:ss */

CONVERT(VARCHAR(19), CONVERT(datetime2, CONCAT(S.RECORD_DATE, ' ', S.RECORD_TIME)), 120)

CONVERT(VARCHAR(19), CAST(TRY_CONVERT(DATE, S.RECORD_DATE) AS DATETIME) + CAST(TRY_CONVERT(TIME, S.RECORD_TIME) AS DATETIME), 120)

    SELECT
        COALESCE(V.Recon_id, S.Recon_id) AS Recon_id,
        COALESCE(V.Claim_Number, S.Claim_Number) AS Claim_Number,
        COALESCE(V.Source_ID, S.Source_ID) AS Source_ID,

        CASE

            WHEN V.V2_value IS NULL OR S.SS_Value IS NULL THEN 'FAIL'
            WHEN UPPER(LTRIM(RTRIM(V.V2_value))) = UPPER(LTRIM(RTRIM(S.SS_Value))) THEN 'PASS'
            ELSE 'FAIL'
        END AS V2_SS_result

    FROM V2 V
    FULL OUTER JOIN SS_ALL_RAW S
        ON S.Source_ID = V.Source_ID
       AND S.Parameter_Name = V.Parameter_Name;

    MERGE audit.NonFinReconciliationresult AS R
    USING
    (
        SELECT Recon_id, Parameter_Name, LOBCode,
               SUM(CASE WHEN V2_value IS NOT NULL THEN 1 ELSE 0 END) AS V2Cnt,
               SUM(CASE WHEN SS_Value IS NOT NULL THEN 1 ELSE 0 END) AS SSCnt,
               SUM(CASE WHEN V2_SS_result = 'PASS' THEN 1 ELSE 0 END) AS MatchCnt,
               SUM(CASE WHEN V2_SS_result = 'FAIL' THEN 1 ELSE 0 END) AS MismatchCnt
        FROM #Compared
        GROUP BY Recon_id, Parameter_Name, LOBCode
    ) AS X
    ON R.Execution_id = @Execution_id
       AND R.Recon_id = X.Recon_id
       AND R.Parameter_Name = X.Parameter_Name
       AND R.LOBCode = X.LOBCode
    WHEN MATCHED THEN
        UPDATE SET R.V2_RecordCount = X.V2Cnt, R.SS_RecordCount_forV2 = X.SSCnt,
                   R.V2_SS_MatchedRecords = X.MatchCnt, R.V2_SS_MismatchedRecords = X.MismatchCnt,
                   R.V2_SS_result = CASE WHEN X.MismatchCnt = 0 THEN 'PASS' ELSE 'FAIL' END
    WHEN NOT MATCHED THEN
        INSERT (Execution_id, Recon_id, Parameter_Name, LOBCode,
                V2_RecordCount, SS_RecordCount_forV2, V2_SS_MatchedRecords, V2_SS_MismatchedRecords, V2_SS_result, Created_at)
        VALUES (@Execution_id, X.Recon_id, X.Parameter_Name, X.LOBCode,
                X.V2Cnt, X.SSCnt, X.MatchCnt, X.MismatchCnt,
                CASE WHEN X.MismatchCnt = 0 THEN 'PASS' ELSE 'FAIL' END, GETDATE());

    MERGE audit.ClaimNonFinReconciliationData AS tgt
    USING (SELECT * FROM #Compared WHERE V2_SS_result = 'FAIL') AS src
    ON tgt.Execution_id = @Execution_id AND tgt.Recon_id = src.Recon_id AND tgt.Source_ID = src.Source_ID
    WHEN MATCHED THEN
        UPDATE SET tgt.V2_value = src.V2_value, tgt.SS_value = src.SS_Value,
                   tgt.V2_SS_result = src.V2_SS_result, tgt.PublicID = src.PublicID
    WHEN NOT MATCHED THEN
        INSERT (Execution_id, Recon_id, Claim_Number, Source_ID, PublicID, LOBCode, Parameter_Name, V2_value, SS_value, V2_SS_result, Created_at)
        VALUES (@Execution_id, src.Recon_id, src.Claim_Number, src.Source_ID, src.PublicID, src.LOBCode, src.Parameter_Name, src.V2_value, src.SS_Value, src.V2_SS_result, GETDATE());

    DROP TABLE #Compared;
END

            ON C.GW_HDR_CASEID = INC.GW_HDR_CASEID AND LTRIM(RTRIM(INC.GCURRENT)) = 'X'
        LEFT JOIN [SourceStaging].VECCASRN.VEC_GW_CIRCS_TYP TYP ON INC.GW_CIRCS_TYPID = TYP.ID
        LEFT JOIN [SourceStaging].[dbo].[MOTOR_CIRC_LOSSCAUSE_MAPPING] MAP
            ON TYP.TYPCODE = MAP.VECTUS_LOSSCAUSE AND C.INCIDENT_CODE = MAP.VECTUS_CIRCUMSTANCE
        WHERE UPPER(LTRIM(RTRIM(C.GCURRENT))) = 'X'
          AND (C.INCIDENT_CODE IS NULL OR UPPER(LTRIM(RTRIM(C.INCIDENT_CODE))) <> 'WINDSCREEN')
          AND NULLIF(LTRIM(RTRIM(C.CLAIM_REF)), '') IS NOT NULL
          AND UPPER(LTRIM(RTRIM(C.PRODUCT))) = 'MOTOR'

        UNION ALL

        SELECT 207, LTRIM(RTRIM(C.CLAIM_REF)), 'HOUSEHOLD', 'LOSS CAUSE',
               LTRIM(RTRIM(C.CLAIM_REF)), 'mig:hh:' + LTRIM(RTRIM(C.CLAIM_REF)),
               CAUSE_CIRC.[cctl_losscause_TYPECODE]
        FROM [SourceStaging].VECCASRN.VEC_GW_CLAIM_SUM C
        INNER JOIN POLICY_HIGHEST_ID P ON P.GW_HDR_CASEID = C.GW_HDR_CASEID AND P.RN = 1
        LEFT JOIN [SourceStaging].VECCASRN.VEC_HH_INC_DETS D ON D.HH_CLAIMID = C.GW_HDR_CASEID
        LEFT JOIN [SourceStaging].VECCASRN.VEC_HH_AXX_CIRCS CIRC_LKP ON D.CIRC = CIRC_LKP.ID
        LEFT JOIN [SourceStaging].VECCASRN.VEC_HH_AXX_CIRCS CAUSE_LKP ON D.CAUSE = CAUSE_LKP.ID
        LEFT JOIN [SourceStaging].[dbo].[CLAIM_LOSSCAUSE_CIRC_MAPPING] CAUSE_CIRC
            ON CAUSE_LKP.CAUSE = CAUSE_CIRC.Vectus_CAUSE AND CIRC_LKP.CIRC = CAUSE_CIRC.Vectus_CIRCUMSTANCE
        WHERE UPPER(LTRIM(RTRIM(C.GCURRENT))) = 'X'
          AND (C.INCIDENT_CODE IS NULL OR UPPER(LTRIM(RTRIM(C.INCIDENT_CODE))) <> 'WINDSCREEN')
          AND NULLIF(LTRIM(RTRIM(C.CLAIM_REF)), '') IS NOT NULL
          AND UPPER(LTRIM(RTRIM(C.PRODUCT))) = 'HOUSEHOLD'

        -- 208 Fault Rating, 211 Vehicle: extend using the same real-transformation
        -- principle once the IS-side target values/tables are confirmed
    ),
    IS_ALL AS
    (
        SELECT 201 AS Recon_id, c.ClaimNumber AS Claim_Number,
               CASE WHEN p.PolicyType = 'PMPersonalMotor' THEN 'MOTOR' ELSE 'HOUSEHOLD' END AS LOBCode,
               'Date of Loss' AS Parameter_Name, c.PublicID,
               CONVERT(VARCHAR(10), c.LossDate, 23) AS IS_Value
        FROM dbo.IS_CLAIM c INNER JOIN dbo.IS_POLICY p ON c.PolicyID = p.PublicID
        WHERE c.PublicID IS NOT NULL

        UNION ALL

        SELECT 203, c.ClaimNumber,
               CASE WHEN p.PolicyType = 'PMPersonalMotor' THEN 'MOTOR' ELSE 'HOUSEHOLD' END,
               'Claim Number', c.PublicID, c.ClaimNumber
        FROM dbo.IS_CLAIM c INNER JOIN dbo.IS_POLICY p ON c.PolicyID = p.PublicID
        WHERE c.PublicID IS NOT NULL

        UNION ALL

        SELECT 206, c.ClaimNumber,
               CASE WHEN p.PolicyType = 'PMPersonalMotor' THEN 'MOTOR' ELSE 'HOUSEHOLD' END,
               'PM Admit Liability Flag', c.PublicID,
               c.InsuredLiabilityView_Adm    -- CONFIRM real column name on dbo.IS_CLAIM
        FROM dbo.IS_CLAIM c INNER JOIN dbo.IS_POLICY p ON c.PolicyID = p.PublicID
        WHERE c.PublicID IS NOT NULL

        UNION ALL

        SELECT 207, c.ClaimNumber,
               CASE WHEN p.PolicyType = 'PMPersonalMotor' THEN 'MOTOR' ELSE 'HOUSEHOLD' END,
               'LOSS CAUSE', c.PublicID, c.LossCause    -- CONFIRM real column name
        FROM dbo.IS_CLAIM c INNER JOIN dbo.IS_POLICY p ON c.PolicyID = p.PublicID
        WHERE c.PublicID IS NOT NULL

        -- 202, 204, 205, 208, 209, 210, 211: extend once real dbo.IS_CLAIM /
        -- dbo.IS_POLICY / dbo.IS_VEHICLE column names are confirmed
    ),
    RawSS AS
    (
        SELECT DISTINCT Source_ID, PublicID, Parameter_Name, SS_value
        FROM audit.ClaimNonFinReconciliationData
        WHERE Execution_id = @Execution_id
    )

    SELECT
        COALESCE(T.Recon_id, I.Recon_id) AS Recon_id,
        COALESCE(T.Claim_Number, I.Claim_Number) AS Claim_Number,
        COALESCE(T.Source_ID, I.PublicID) AS Source_ID,
        COALESCE(T.PublicID, I.PublicID) AS PublicID,
        COALESCE(T.LOBCode, I.LOBCode) AS LOBCode,
        COALESCE(T.Parameter_Name, I.Parameter_Name) AS Parameter_Name,
        R.SS_value AS SS_Value_Raw,
        T.SS_Value AS SS_Value_Transformed,
        I.IS_Value,
        CASE
            WHEN T.SS_Value IS NULL AND I.IS_Value IS NULL THEN 'PASS'
            WHEN T.SS_Value IS NULL OR I.IS_Value IS NULL THEN 'FAIL'
            WHEN UPPER(LTRIM(RTRIM(T.SS_Value))) = UPPER(LTRIM(RTRIM(I.IS_Value))) THEN 'PASS'
            ELSE 'FAIL'
        END AS SS_IS_result
    INTO #Compared2
    FROM SS_ALL_TRANSFORMED T
    FULL OUTER JOIN IS_ALL I ON I.PublicID = T.PublicID AND I.Parameter_Name = T.Parameter_Name
    LEFT JOIN RawSS R ON R.PublicID = COALESCE(T.PublicID, I.PublicID) AND R.Parameter_Name = COALESCE(T.Parameter_Name, I.Parameter_Name);

    MERGE audit.NonFinReconciliationresult AS R
    USING
    (
        SELECT Recon_id, Parameter_Name, LOBCode,
               SUM(CASE WHEN SS_Value_Transformed IS NOT NULL THEN 1 ELSE 0 END) AS SSCnt,
               SUM(CASE WHEN IS_Value IS NOT NULL THEN 1 ELSE 0 END) AS ISCnt,
               SUM(CASE WHEN SS_IS_result = 'PASS' THEN 1 ELSE 0 END) AS MatchCnt,
               SUM(CASE WHEN SS_IS_result = 'FAIL' THEN 1 ELSE 0 END) AS MismatchCnt
        FROM #Compared2
        GROUP BY Recon_id, Parameter_Name, LOBCode
    ) AS X
    ON R.Execution_id = @Execution_id
       AND R.Recon_id = X.Recon_id
       AND R.Parameter_Name = X.Parameter_Name
       AND R.LOBCode = X.LOBCode
    WHEN MATCHED THEN
        UPDATE SET R.SS_RecordCount_forIS = X.SSCnt, R.IS_RecordCount = X.ISCnt,
                   R.SS_IS_MatchedRecords = X.MatchCnt, R.SS_IS_MismatchedRecords = X.MismatchCnt,
                   R.SS_IS_result = CASE WHEN X.MismatchCnt = 0 THEN 'PASS' ELSE 'FAIL' END
    WHEN NOT MATCHED THEN
        INSERT (Execution_id, Recon_id, Parameter_Name, LOBCode,
                SS_RecordCount_forIS, IS_RecordCount, SS_IS_MatchedRecords, SS_IS_MismatchedRecords, SS_IS_result, Created_at)
        VALUES (@Execution_id, X.Recon_id, X.Parameter_Name, X.LOBCode,
                X.SSCnt, X.ISCnt, X.MatchCnt, X.MismatchCnt,
                CASE WHEN X.MismatchCnt = 0 THEN 'PASS' ELSE 'FAIL' END, GETDATE());

    MERGE audit.ClaimNonFinReconciliationData AS tgt
    USING (SELECT * FROM #Compared2 WHERE SS_IS_result = 'FAIL') AS src
    ON tgt.Execution_id = @Execution_id AND tgt.Recon_id = src.Recon_id AND tgt.PublicID = src.PublicID
    WHEN MATCHED THEN
        UPDATE SET tgt.SS_Value_Transformed = src.SS_Value_Transformed, tgt.IS_value = src.IS_Value, tgt.SS_IS_result = src.SS_IS_result
    WHEN NOT MATCHED THEN
        INSERT (Execution_id, Recon_id, Claim_Number, Source_ID, PublicID, LOBCode, Parameter_Name, SS_Value_Transformed, IS_value, SS_IS_result, Created_at)
        VALUES (@Execution_id, src.Recon_id, src.Claim_Number, src.Source_ID, src.PublicID, src.LOBCode, src.Parameter_Name, src.SS_Value_Transformed, src.IS_Value, src.SS_IS_result, GETDATE());

    DROP TABLE #Compared2;
END




INSERT INTO audit.NonFinReconciliationresult
    (Execution_id, Recon_id, Parameter_Name, LOBCode,
     V2_RecordCount, SS_RecordCount_forV2, V2_SS_MatchedRecords, V2_SS_MismatchedRecords, V2_SS_result, Created_at)
SELECT
    @Execution_id, Recon_id, Parameter_Name, LOBCode,
    SUM(CASE WHEN V2_value IS NOT NULL THEN 1 ELSE 0 END),
    SUM(CASE WHEN SS_Value IS NOT NULL THEN 1 ELSE 0 END),
    SUM(CASE WHEN V2_SS_result = 'PASS' THEN 1 ELSE 0 END),
    SUM(CASE WHEN V2_SS_result = 'FAIL' THEN 1 ELSE 0 END),
    CASE WHEN SUM(CASE WHEN V2_SS_result = 'FAIL' THEN 1 ELSE 0 END) = 0 THEN 'PASS' ELSE 'FAIL' END,
    GETDATE()
FROM #Compared
GROUP BY Recon_id, Parameter_Name, LOBCode;


UPDATE R
SET R.SS_RecordCount_forIS = X.SSCnt,
    R.IS_RecordCount = X.ISCnt,
    R.SS_IS_MatchedRecords = X.MatchCnt,
    R.SS_IS_MismatchedRecords = X.MismatchCnt,
    R.SS_IS_result = CASE WHEN X.MismatchCnt = 0 THEN 'PASS' ELSE 'FAIL' END
FROM audit.NonFinReconciliationresult R
INNER JOIN
(
    SELECT Recon_id, Parameter_Name, LOBCode,
           SUM(CASE WHEN SS_Value_Transformed IS NOT NULL THEN 1 ELSE 0 END) AS SSCnt,
           SUM(CASE WHEN IS_Value IS NOT NULL THEN 1 ELSE 0 END) AS ISCnt,
           SUM(CASE WHEN SS_IS_result = 'PASS' THEN 1 ELSE 0 END) AS MatchCnt,
           SUM(CASE WHEN SS_IS_result = 'FAIL' THEN 1 ELSE 0 END) AS MismatchCnt
    FROM #Compared2
    GROUP BY Recon_id, Parameter_Name, LOBCode
) X ON X.Recon_id = R.Recon_id AND X.Parameter_Name = R.Parameter_Name AND X.LOBCode = R.LOBCode
WHERE R.Execution_id = @Execution_id;
