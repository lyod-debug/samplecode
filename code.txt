;WITH POLICY_HIGHEST_ID AS
(
    SELECT
        P.GW_HDR_CASEID, P.POLICY_REF, P.ID, P.PRODUCT, P.POLICY_BRAND,
        P.CUR_INCEP_DATE, P.POL_INCEP_DATE,
        ROW_NUMBER() OVER (PARTITION BY P.GW_HDR_CASEID ORDER BY P.ID DESC) AS RN
    FROM [SourceStaging].VECCASRN.VEC_GWP_POLICY P
),
SS_ALL AS
(
    -- 201: Date of Loss
    SELECT 201 AS Recon_id, LTRIM(RTRIM(C.CLAIM_REF)) AS Claim_Number,
           UPPER(LTRIM(RTRIM(C.PRODUCT))) AS LOBCode, 'Date of Loss' AS Parameter_Name,
           CONVERT(VARCHAR(10), C.ACTUAL_DOL, 23) AS SS_Value
    FROM [SourceStaging].VECCASRN.VEC_GW_CLAIM_SUM C
    INNER JOIN POLICY_HIGHEST_ID P ON P.GW_HDR_CASEID = C.GW_HDR_CASEID AND P.RN = 1
    WHERE UPPER(LTRIM(RTRIM(C.GCURRENT))) = 'X'
      AND (C.INCIDENT_CODE IS NULL OR UPPER(LTRIM(RTRIM(C.INCIDENT_CODE))) <> 'WINDSCREEN')
      AND NULLIF(LTRIM(RTRIM(C.CLAIM_REF)), '') IS NOT NULL

    UNION ALL

    -- 202: Reported Date
    SELECT 202, LTRIM(RTRIM(C.CLAIM_REF)), UPPER(LTRIM(RTRIM(C.PRODUCT))), 'Reported Date',
           CONVERT(VARCHAR(10), C.REPORTED_DATE, 23)
    FROM [SourceStaging].VECCASRN.VEC_GW_CLAIM_SUM C
    INNER JOIN POLICY_HIGHEST_ID P ON P.GW_HDR_CASEID = C.GW_HDR_CASEID AND P.RN = 1
    WHERE UPPER(LTRIM(RTRIM(C.GCURRENT))) = 'X'
      AND (C.INCIDENT_CODE IS NULL OR UPPER(LTRIM(RTRIM(C.INCIDENT_CODE))) <> 'WINDSCREEN')
      AND NULLIF(LTRIM(RTRIM(C.CLAIM_REF)), '') IS NOT NULL

    UNION ALL

    -- 203: Claim Number
    SELECT 203, LTRIM(RTRIM(C.CLAIM_REF)), UPPER(LTRIM(RTRIM(C.PRODUCT))), 'Claim Number',
           LTRIM(RTRIM(C.CLAIM_REF))
    FROM [SourceStaging].VECCASRN.VEC_GW_CLAIM_SUM C
    INNER JOIN POLICY_HIGHEST_ID P ON P.GW_HDR_CASEID = C.GW_HDR_CASEID AND P.RN = 1
    WHERE UPPER(LTRIM(RTRIM(C.GCURRENT))) = 'X'
      AND (C.INCIDENT_CODE IS NULL OR UPPER(LTRIM(RTRIM(C.INCIDENT_CODE))) <> 'WINDSCREEN')
      AND NULLIF(LTRIM(RTRIM(C.CLAIM_REF)), '') IS NOT NULL

    UNION ALL

    -- 204: No Claim Bonus Status (Motor only)
    SELECT 204, LTRIM(RTRIM(C.CLAIM_REF)), UPPER(LTRIM(RTRIM(C.PRODUCT))), 'No Claim Bonus Status',
           LTRIM(RTRIM(C.NCB_STATUS))
    FROM [SourceStaging].VECCASRN.VEC_GW_CLAIM_SUM C
    INNER JOIN POLICY_HIGHEST_ID P ON P.GW_HDR_CASEID = C.GW_HDR_CASEID AND P.RN = 1
    WHERE UPPER(LTRIM(RTRIM(C.GCURRENT))) = 'X'
      AND (C.INCIDENT_CODE IS NULL OR UPPER(LTRIM(RTRIM(C.INCIDENT_CODE))) <> 'WINDSCREEN')
      AND NULLIF(LTRIM(RTRIM(C.CLAIM_REF)), '') IS NOT NULL
      AND UPPER(LTRIM(RTRIM(C.PRODUCT))) = 'MOTOR'

    UNION ALL

    -- 205: Claim Finalised Date
    SELECT 205, LTRIM(RTRIM(C.CLAIM_REF)), UPPER(LTRIM(RTRIM(C.PRODUCT))), 'Claim finalised date',
           CAST(CASE WHEN S.RECORD_DATE IS NOT NULL AND S.RECORD_TIME IS NOT NULL
                     THEN CONVERT(VARCHAR(10), S.RECORD_DATE, 23) + ' ' + CONVERT(VARCHAR(8), S.RECORD_TIME, 108)
                END AS VARCHAR(255))
    FROM [SourceStaging].VECCASRN.VEC_GW_CLAIM_SUM C
    INNER JOIN POLICY_HIGHEST_ID P ON P.GW_HDR_CASEID = C.GW_HDR_CASEID AND P.RN = 1
    LEFT JOIN [SourceStaging].VECCASRN.VEC_GW_CASE_STATUS S
        ON S.CASEID = C.GW_HDR_CASEID
       AND UPPER(LTRIM(RTRIM(S.STATUS))) = 'FINALISED'
       AND LTRIM(RTRIM(S.GCURRENT)) = 'X'
    WHERE UPPER(LTRIM(RTRIM(C.CLAIM_STATUS))) = 'F'
      AND UPPER(LTRIM(RTRIM(C.GCURRENT))) = 'X'
      AND (C.INCIDENT_CODE IS NULL OR UPPER(LTRIM(RTRIM(C.INCIDENT_CODE))) <> 'WINDSCREEN')
      AND NULLIF(LTRIM(RTRIM(C.CLAIM_REF)), '') IS NOT NULL

    UNION ALL

    -- 206: PM Admit Liability Flag (Motor only, concatenated raw pair)
    SELECT 206, LTRIM(RTRIM(C.CLAIM_REF)), 'MOTOR', 'PM Admit Liability Flag',
           CAST(COALESCE(LTRIM(RTRIM(INC.INSDADMITLIAB)), '#NULL#') + '|' +
                COALESCE(LTRIM(RTRIM(REF.LIABILITY)), '#NULL#') AS VARCHAR(255))
    FROM [SourceStaging].VECCASRN.VEC_GW_CLAIM_SUM C
    INNER JOIN POLICY_HIGHEST_ID P ON P.GW_HDR_CASEID = C.GW_HDR_CASEID AND P.RN = 1
    LEFT JOIN [SourceStaging].VECCASRN.VEC_GW_CIRCS_INC INC
        ON C.GW_HDR_CASEID = INC.GW_HDR_CASEID AND LTRIM(RTRIM(INC.GCURRENT)) = 'X'
    LEFT JOIN [SourceStaging].VECCASRN.VEC_GW_CIRCS_REF REF ON REF.ID = INC.GW_CIRCS_REFID
    WHERE UPPER(LTRIM(RTRIM(C.GCURRENT))) = 'X'
      AND (C.INCIDENT_CODE IS NULL OR UPPER(LTRIM(RTRIM(C.INCIDENT_CODE))) <> 'WINDSCREEN')
      AND NULLIF(LTRIM(RTRIM(C.CLAIM_REF)), '') IS NOT NULL
      AND UPPER(LTRIM(RTRIM(C.PRODUCT))) = 'MOTOR'

    UNION ALL

    -- 207: Loss Cause (Household)
    SELECT 207, LTRIM(RTRIM(C.CLAIM_REF)), 'HOUSEHOLD', 'LOSS CAUSE',
           CAST(COALESCE(LTRIM(RTRIM(CAUSE_LKP.CAUSE)), '#NULL#') + '|' +
                COALESCE(LTRIM(RTRIM(CIRC_LKP.CIRC)), '#NULL#') AS VARCHAR(255))
    FROM [SourceStaging].VECCASRN.VEC_GW_CLAIM_SUM C
    INNER JOIN POLICY_HIGHEST_ID P ON P.GW_HDR_CASEID = C.GW_HDR_CASEID AND P.RN = 1
    LEFT JOIN [SourceStaging].VECCASRN.VEC_HH_INC_DETS D ON D.HH_CLAIMID = C.GW_HDR_CASEID
    LEFT JOIN [SourceStaging].VECCASRN.VEC_HH_AXX_CIRCS CIRC_LKP ON D.CIRC = CIRC_LKP.ID
    LEFT JOIN [SourceStaging].VECCASRN.VEC_HH_AXX_CIRCS CAUSE_LKP ON D.CAUSE = CAUSE_LKP.ID
    WHERE UPPER(LTRIM(RTRIM(C.GCURRENT))) = 'X'
      AND (C.INCIDENT_CODE IS NULL OR UPPER(LTRIM(RTRIM(C.INCIDENT_CODE))) <> 'WINDSCREEN')
      AND NULLIF(LTRIM(RTRIM(C.CLAIM_REF)), '') IS NOT NULL
      AND UPPER(LTRIM(RTRIM(C.PRODUCT))) = 'HOUSEHOLD'

    UNION ALL

    -- 207: Loss Cause (Motor)
    SELECT 207, LTRIM(RTRIM(C.CLAIM_REF)), 'MOTOR', 'LOSS CAUSE',
           CAST(COALESCE(LTRIM(RTRIM(TYP.TYPCODE)), '#NULL#') + '|' +
                COALESCE(LTRIM(RTRIM(C.INCIDENT_CODE)), '#NULL#') AS VARCHAR(255))
    FROM [SourceStaging].VECCASRN.VEC_GW_CLAIM_SUM C
    INNER JOIN POLICY_HIGHEST_ID P ON P.GW_HDR_CASEID = C.GW_HDR_CASEID AND P.RN = 1
    LEFT JOIN [SourceStaging].VECCASRN.VEC_GW_CIRCS_INC INC
        ON C.GW_HDR_CASEID = INC.GW_HDR_CASEID AND LTRIM(RTRIM(INC.GCURRENT)) = 'X'
    LEFT JOIN [SourceStaging].VECCASRN.VEC_GW_CIRCS_TYP TYP ON INC.GW_CIRCS_TYPID = TYP.ID
    WHERE UPPER(LTRIM(RTRIM(C.GCURRENT))) = 'X'
      AND (C.INCIDENT_CODE IS NULL OR UPPER(LTRIM(RTRIM(C.INCIDENT_CODE))) <> 'WINDSCREEN')
      AND NULLIF(LTRIM(RTRIM(C.CLAIM_REF)), '') IS NOT NULL
      AND UPPER(LTRIM(RTRIM(C.PRODUCT))) = 'MOTOR'

    UNION ALL

    -- 208: Fault Rating (Motor only, same raw pair as 206)
    SELECT 208, LTRIM(RTRIM(C.CLAIM_REF)), 'MOTOR', 'Fault Rating',
           CAST(COALESCE(LTRIM(RTRIM(INC.INSDADMITLIAB)), '#NULL#') + '|' +
                COALESCE(LTRIM(RTRIM(REF.LIABILITY)), '#NULL#') AS VARCHAR(255))
    FROM [SourceStaging].VECCASRN.VEC_GW_CLAIM_SUM C
    INNER JOIN POLICY_HIGHEST_ID P ON P.GW_HDR_CASEID = C.GW_HDR_CASEID AND P.RN = 1
    LEFT JOIN [SourceStaging].VECCASRN.VEC_GW_CIRCS_INC INC
        ON C.GW_HDR_CASEID = INC.GW_HDR_CASEID AND LTRIM(RTRIM(INC.GCURRENT)) = 'X'
    LEFT JOIN [SourceStaging].VECCASRN.VEC_GW_CIRCS_REF REF ON REF.ID = INC.GW_CIRCS_REFID
    WHERE UPPER(LTRIM(RTRIM(C.GCURRENT))) = 'X'
      AND (C.INCIDENT_CODE IS NULL OR UPPER(LTRIM(RTRIM(C.INCIDENT_CODE))) <> 'WINDSCREEN')
      AND NULLIF(LTRIM(RTRIM(C.CLAIM_REF)), '') IS NOT NULL
      AND UPPER(LTRIM(RTRIM(C.PRODUCT))) = 'MOTOR'

    UNION ALL

    -- 209: Policy Number
    SELECT 209, LTRIM(RTRIM(C.CLAIM_REF)), UPPER(LTRIM(RTRIM(C.PRODUCT))), 'Policy Number',
           LTRIM(RTRIM(P.POLICY_REF))
    FROM [SourceStaging].VECCASRN.VEC_GW_CLAIM_SUM C
    INNER JOIN POLICY_HIGHEST_ID P ON P.GW_HDR_CASEID = C.GW_HDR_CASEID AND P.RN = 1
    WHERE UPPER(LTRIM(RTRIM(C.GCURRENT))) = 'X'
      AND (C.INCIDENT_CODE IS NULL OR UPPER(LTRIM(RTRIM(C.INCIDENT_CODE))) <> 'WINDSCREEN')
      AND NULLIF(LTRIM(RTRIM(C.CLAIM_REF)), '') IS NOT NULL

    UNION ALL

    -- 210: Policy Term Inception Date
    SELECT 210, LTRIM(RTRIM(C.CLAIM_REF)), UPPER(LTRIM(RTRIM(C.PRODUCT))), 'Policy Term Inception Date',
           CONVERT(VARCHAR(10), P.POL_INCEP_DATE, 23)
    FROM [SourceStaging].VECCASRN.VEC_GW_CLAIM_SUM C
    INNER JOIN POLICY_HIGHEST_ID P ON P.GW_HDR_CASEID = C.GW_HDR_CASEID AND P.RN = 1
    WHERE UPPER(LTRIM(RTRIM(C.GCURRENT))) = 'X'
      AND (C.INCIDENT_CODE IS NULL OR UPPER(LTRIM(RTRIM(C.INCIDENT_CODE))) <> 'WINDSCREEN')
      AND NULLIF(LTRIM(RTRIM(C.CLAIM_REF)), '') IS NOT NULL
)
