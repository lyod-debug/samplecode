;WITH POLICY_HIGHEST_ID AS
(
    SELECT
        P.GW_HDR_CASEID,
        P.ID,
        P.PRODUCT,
        ROW_NUMBER() OVER (PARTITION BY P.GW_HDR_CASEID ORDER BY P.ID DESC) AS RN
    FROM [SourceStaging].VECCASRN.VEC_GWP_POLICY P
)
Insert into [audit].[Tmp_Attr_Recon]
(Matrixname, recon_ID, LOB, SS_IS_Match, SS_IS_MisMatch) values
(
    'LossCause'
    ,207
    ,'MOTOR'

    ,( Select count(*) from (
        Select C.CLAIM_REF, CAUSE.[cctl_losscause_typecode] AS LossCause
        FROM [SourceStaging].VECCASRN.VEC_GW_CLAIM_SUM C
        INNER JOIN POLICY_HIGHEST_ID P ON P.GW_HDR_CASEID = C.GW_HDR_CASEID AND P.RN = 1
        LEFT JOIN [SourceStaging].VECCASRN.VEC_GW_CIRCS_INC INC
            ON C.GW_HDR_CASEID = INC.GW_HDR_CASEID AND LTRIM(RTRIM(INC.GCURRENT)) = 'X'
        LEFT JOIN [SourceStaging].VECCASRN.VEC_GW_CIRCS_TYP TYP
            ON INC.GW_CIRCS_TYPID = TYP.ID
        LEFT JOIN [SourceStaging].[dbo].[MOTOR_CIRC_LOSSCAUSE_MAPPING] CAUSE
            ON TYP.TYPCODE      = CAUSE.VECTUS_LOSSCAUSE
           AND C.INCIDENT_CODE  = CAUSE.VECTUS_CIRCUMSTANCE
        where LTRIM(RTRIM(C.GCURRENT)) = 'X'
          and UPPER(LTRIM(RTRIM(P.PRODUCT))) = 'MOTOR'
          AND COALESCE(NULLIF(UPPER(LTRIM(RTRIM(C.INCIDENT_CODE))), ''), '#NULL#') <> 'WINDSCREEN'
          AND NULLIF(LTRIM(RTRIM(C.CLAIM_REF)), '') IS NOT NULL

        INTERSECT

        SELECT c.ClaimNumber, c.LossCause      -- CONFIRM real column name on dbo.IS_CLAIM
            FROM dbo.IS_CLAIM c
                JOIN dbo.IS_POLICY p on c.PolicyID = p.PUBLICID
            WHERE c.PublicID IS NOT NULL
              and POLICYTYPE = 'PMPersonalMotor'
        ) A
    )

    ,( Select count(*) from (
        Select C.CLAIM_REF, CAUSE.[cctl_losscause_typecode] AS LossCause
        FROM [SourceStaging].VECCASRN.VEC_GW_CLAIM_SUM C
        INNER JOIN POLICY_HIGHEST_ID P ON P.GW_HDR_CASEID = C.GW_HDR_CASEID AND P.RN = 1
        LEFT JOIN [SourceStaging].VECCASRN.VEC_GW_CIRCS_INC INC
            ON C.GW_HDR_CASEID = INC.GW_HDR_CASEID AND LTRIM(RTRIM(INC.GCURRENT)) = 'X'
        LEFT JOIN [SourceStaging].VECCASRN.VEC_GW_CIRCS_TYP TYP
            ON INC.GW_CIRCS_TYPID = TYP.ID
        LEFT JOIN [SourceStaging].[dbo].[MOTOR_CIRC_LOSSCAUSE_MAPPING] CAUSE
            ON TYP.TYPCODE      = CAUSE.VECTUS_LOSSCAUSE
           AND C.INCIDENT_CODE  = CAUSE.VECTUS_CIRCUMSTANCE
        where LTRIM(RTRIM(C.GCURRENT)) = 'X'
          and UPPER(LTRIM(RTRIM(P.PRODUCT))) = 'MOTOR'
          AND COALESCE(NULLIF(UPPER(LTRIM(RTRIM(C.INCIDENT_CODE))), ''), '#NULL#') <> 'WINDSCREEN'
          AND NULLIF(LTRIM(RTRIM(C.CLAIM_REF)), '') IS NOT NULL

        EXCEPT

        SELECT c.ClaimNumber, c.LossCause
            FROM dbo.IS_CLAIM c
                JOIN dbo.IS_POLICY p on c.PolicyID = p.PUBLICID
            WHERE c.PublicID IS NOT NULL
              and POLICYTYPE = 'PMPersonalMotor'
        ) B
    )
)
