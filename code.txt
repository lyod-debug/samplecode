   property/motor risk unit procs (per your note: not every cc_riskunit
   column is in migration scope).

   TYPE NOTE: cc_riskunit's PolicyID/VehicleID/PolicyLocationID/
   VehicleLocationID are bigint FKs on the CC side, but in this staging
   layer they hold PublicID strings (same pattern as every other FK in
   this pipeline - 'mig:...' style) which the bulk-load utility resolves

CREATE OR ALTER PROCEDURE [dbo].[usp_Load_IS_VEHICLE_RISKUNIT_MOTOR]
AS
BEGIN
TRUNCATE TABLE IntermediateStaging_DEV.dbo.IS_VEHICLE_RISKUNIT_MOTOR;

INSERT INTO IntermediateStaging_DEV.dbo.IS_VEHICLE_RISKUNIT_MOTOR
(
    CLAIM_REF, RiskUnitPublicID, RUNumber, PolicyID, VehicleLocationID,
    SubPolicyID_Adm, ClaimsRiskType_Adm, CourtesyCar_Adm, CoverableType_Adm, CoverType_Adm,
    CriminalConvictions_Adm, CubicCapacity_Adm, EffectiveDate_Adm, ExpirationDate_Adm,
    NotOnCover_Adm, RegisteredOwnerName_Adm, SecurityFeatures_Adm, SignWriting_Adm,
    TempAddedExpDate_Adm, TempAdded_Adm, TempAddedReason_Adm, TelematicsStdDis_Adm,
    TelematicsFlag_Adm, TelematicsInstallationDate_Adm, TelematicsProduct_Adm,
    TelematicsStatus_Adm, Transmission_Adm, TierType_Adm, VehicleID, VehicleOwner_Adm,
    VehicleOwnerName_Adm
)
SELECT
    CLM.CLAIM_REF,
    /* deterministic PublicID, not ROW_NUMBER() */
    'mig:motorriskunit' + CONVERT(VARCHAR(64), CLM.ID) + '_' + CONVERT(VARCHAR(64), SR.ID),
    /* RUNumber: real sequential int, RU.PUBLICID is a string - never put that here */
    ROW_NUMBER() OVER (PARTITION BY SR.GWP_POLICYID ORDER BY RU.ID),
    /* PolicyID: FIX - must equal IS_POLICY.PublicID, not raw source Policy_ID */
    CLM.PUBLICID,
    NULL,   -- VehicleLocationID: still needs IS_POLICYLOCATION's public-id formula
    SR.SUB_POLICYID,
    TRT.GW_TypeCode,
    CASE WHEN LM.LEND_METHOD='Courtesy Car' THEN 1 ELSE NULL END,
    CVBT.GW_TypeCode,
    CVT.GW_TypeCode,
    CASE WHEN RU.CONVICTIONS='Y' THEN 1 ELSE NULL END,
    V.CUBIC_CAPACITY,
    P.EffectiveDate,

    CASE WHEN DOC.CVRD_ON_VEH='Y' THEN 1 ELSE NULL END,
    V.REGOWNER_NAME,
    SEC.GW_TypeCode,
    V.SIGNWRITING,
    CONVERT(varchar(19), V.TAV_DATE, 120),
    CASE WHEN V.TAV_FLAG='Y' THEN 1 ELSE NULL END,
    TAVRSN.GW_TypeCode,
    TSTD.GW_TypeCode,
    CASE WHEN V.TELEM_FLAG='Y' THEN 1 ELSE NULL END,
    CONVERT(varchar(19), V.TELEM_INSTALL, 120),
    TPROD.GW_TypeCode,
    TSTAT.GW_TypeCode,
    V.TRANSMISSION,
    TIER.GW_TypeCode,
    /* VehicleID: FIX - was hard-coded NULL, this broke your recon.
       Matches the PublicID formula usp_Load_SS_to_IS_VEHICLE uses for
       the AD/1st-party vehicle off the same VEC_GWP_VEHICLE row. */
    'mig:advehicle' + CONVERT(VARCHAR(64), V.ID),
    CASE WHEN V.VEH_OWNER='Y' THEN 1 ELSE NULL END,
    V.VEH_OWNER_NAME
FROM IntermediateStaging_DEV.dbo.IS_CLAIM_MASTER CLM
INNER JOIN IntermediateStaging_DEV.dbo.IS_POLICY P ON P.PublicID = CLM.PUBLICID
INNER JOIN [SourceStaging].[VECCASRN].VEC_GWP_SLCTD_RISK SR ON CLM.GW_HDR_CASEID = SR.GW_HDR_CASEID

/* =====================================================================
   VALIDATION QUERIES — run after all three procs, before CCST copy
   ===================================================================== */

-- 1) Any NOT NULL business column actually NULL (mirrors GW's own check,
--    catch it here instead of at integrity-check time)
SELECT PublicID
FROM IntermediateStaging_DEV.dbo.IS_RISKUNIT
WHERE PublicID IS NULL OR RUNumber IS NULL OR PolicyID IS NULL
   OR CoverableType_Adm IS NULL OR Subtype IS NULL;

-- 2) PublicID uniqueness
SELECT PublicID, COUNT(*) FROM IntermediateStaging_DEV.dbo.IS_RISKUNIT
GROUP BY PublicID HAVING COUNT(*) > 1;

-- 3) PolicyID must resolve to a real IS_POLICY row (this is the FK fix -
--    confirm zero rows after applying it)
SELECT RU.PublicID, RU.PolicyID
FROM IntermediateStaging_DEV.dbo.IS_RISKUNIT RU
LEFT JOIN IntermediateStaging_DEV.dbo.IS_POLICY POL ON POL.PublicID = RU.PolicyID
WHERE POL.PublicID IS NULL;

-- 4) Motor: VehicleID must resolve to a real IS_VEHICLE row
SELECT RU.PublicID, RU.VehicleID
FROM IntermediateStaging_DEV.dbo.IS_RISKUNIT RU
LEFT JOIN IntermediateStaging_DEV.dbo.IS_VEHICLE VEHI ON VEHI.PublicID = RU.VehicleID
WHERE RU.Subtype = 'VehicleRU' AND (RU.VehicleID IS NULL OR VEHI.PublicID IS NULL);

-- 5) Unmapped CoverableType_Adm codes on the source side (would surface
--    as NULL -> NOT NULL violation) - check both subtypes
SELECT DISTINCT RSK.COVERABLE_TYPE AS unmapped_household_coverabletype
FROM SourceStaging.VECCASRN.VEC_GWP_RISKUNIT RSK
WHERE RSK.COVERABLE_TYPE IS NOT NULL
  AND NOT EXISTS (
      SELECT 1 FROM SourceStaging.dbo.TYPELIST_TABLE_MAPPING TM
      WHERE TM.Vectus_TypeCode = RSK.COVERABLE_TYPE AND TM.TypeList_Name = 'CoverableType_Adm'
  );

-- 6) RUNumber sanity - confirm it's actually numeric-only, no leftover
--    string PublicID slipping through
SELECT PublicID, RUNumber
FROM IntermediateStaging_DEV.dbo.IS_RISKUNIT
WHERE TRY_CONVERT(int, RUNumber) IS NULL;

-- 7) Row-count parity between the two typed tables and consolidated table
SELECT
    (SELECT COUNT(*) FROM IntermediateStaging_DEV.dbo.IS_PROPERTY_RISKUNIT_HOUSEHOLD) AS property_source_cnt,
    (SELECT COUNT(*) FROM IntermediateStaging_DEV.dbo.IS_VEHICLE_RISKUNIT_MOTOR)       AS motor_source_cnt,
    (SELECT COUNT(*) FROM IntermediateStaging_DEV.dbo.IS_RISKUNIT WHERE Subtype='PropertyRU') AS property_final_cnt,
    (SELECT COUNT(*) FROM IntermediateStaging_DEV.dbo.IS_RISKUNIT WHERE Subtype='VehicleRU')  AS motor_final_cnt;
