CREATE OR ALTER PROCEDURE [dbo].[usp_Load_IS_EXPOSURE_MOTOR]
AS
BEGIN

TRUNCATE TABLE IntermediateStaging_DEV.dbo.IS_EXPOSURE_MOTOR;

/* --------------------------------------------------------------
   BLOCK A: 1st Party AD Vehicle Exposures
   FIXED: CLAIM_REF IS NOT NULL added to the LKP_Dupe_MotorAD check.
   STILL EXCLUDES 'Migrate-group' entirely - open BA question.
   -------------------------------------------------------------- */
;WITH AD_ELIGIBLE AS (
    SELECT AD.ID, AD.GW_HDR_CASEID
    FROM SourceStaging.VECCASRN.VEC_GW_MOTOR_AD AD
    WHERE NOT EXISTS (
        SELECT 1 FROM SourceStaging.dbo.LKP_Dupe_MotorAD DUPE
        WHERE DUPE.CLAIM_REF IS NOT NULL
          AND (DUPE.Solution LIKE 'Descope%' OR DUPE.Solution LIKE 'Migrate-group%')
          AND AD.ID = DUPE.GW_MOTOR_AD_ID
    )
),
AD_CANDIDATES AS (
    SELECT
        CLM.CLAIM_REF, CLM.PUBLICID AS ClaimPublicID, CLM.GW_HDR_CASEID,
        AD.ID AS AD_ID, CIRC.GW_CIRCS_TYPID, RSK.COVERABLE_TYPE
    FROM IntermediateStaging_DEV.dbo.IS_CLAIM_MASTER CLM
    INNER JOIN AD_ELIGIBLE AD ON AD.GW_HDR_CASEID = CLM.GW_HDR_CASEID
    INNER JOIN SourceStaging.VECCASRN.VEC_GW_VEHICLE VEH ON VEH.CASEID = AD.GW_HDR_CASEID
    LEFT JOIN SourceStaging.VECCASRN.VEC_GW_CIRCS_INC CIRC
        ON CIRC.GW_HDR_CASEID = CLM.GW_HDR_CASEID AND CIRC.GCURRENT = 'X'
    INNER JOIN SourceStaging.VECCASRN.VEC_GWP_SLCTD_RISK SR ON SR.GW_HDR_CASEID = CLM.GW_HDR_CASEID
    INNER JOIN SourceStaging.VECCASRN.VEC_GWP_RISKUNIT RSK ON RSK.GWP_POLICYID = SR.GWP_POLICYID AND RSK.PUBLICID = SR.PUBLICID
    WHERE UPPER(LTRIM(RTRIM(CLM.PRODUCT))) = 'MOTOR'
      AND NOT (RSK.COVER_TYPE = 'tpo')
      AND NOT (RSK.COVER_TYPE = 'tpft' AND COALESCE(CIRC.GW_CIRCS_TYPID, 0) <> 5)
)
INSERT INTO IntermediateStaging_DEV.dbo.IS_EXPOSURE_MOTOR
(PublicID, ClaimID, VectusCaseID_Adm, GW_HDR_CASEID, SourceOrigin_Adm, ExposureType, PrimaryCoverage,
 CoverageSubType, LossParty, ClaimOrder, CreateTime, CLAIM_REF, ClaimantType)
SELECT
    'mig:exp:ad:' + AC.ClaimPublicID + '_' + CONVERT(VARCHAR(64), AC.AD_ID),
    AC.ClaimPublicID, CONVERT(VARCHAR(64), AC.AD_ID), AC.GW_HDR_CASEID, 'AD',
    R.ExposureType, R.PrimaryCoverage, R.CoverageSubType, R.LossParty,
    0,
    (SELECT CONVERT(DATETIME2(7), CONCAT(VC.CREATE_DATE, ' ', VC.CREATE_TIME))
     FROM SourceStaging.VECCASRN.VEC_CASE VC WHERE VC.ID = AC.AD_ID),
    AC.CLAIM_REF, 'insured'
FROM AD_CANDIDATES AC
INNER JOIN dbo.LKP_EXPOSURE_MOTOR_RULES R ON R.RuleKey =
    CASE
        WHEN AC.COVERABLE_TYPE = 'PMVan' AND AC.GW_CIRCS_TYPID = 5 THEN 'AD_VAN_THEFT'
        WHEN AC.COVERABLE_TYPE = 'PMVan' THEN 'AD_VAN'
        WHEN AC.GW_CIRCS_TYPID = 5 THEN 'AD_THEFT'
        ELSE 'AD_STD'
    END;


/* --------------------------------------------------------------
   BLOCK B: 3rd Party Exposures (Vehicle / Injury / Property)
   -------------------------------------------------------------- */
;WITH TP_BASE AS (
    SELECT
        CLM.CLAIM_REF, CLM.PUBLICID AS ClaimPublicID, CLM.GW_HDR_CASEID, TP.ID AS TP_ID,
        TPS.VEHICLE, TPS.INJURY, TPS.PROPERTY, TPTYPE.DISPLAY_STRING AS ClaimantRoleDesc
    FROM IntermediateStaging_DEV.dbo.IS_CLAIM_MASTER CLM
    INNER JOIN SourceStaging.VECCASRN.VEC_GW_MOTOR_TP TP ON TP.GW_HDR_CASEID = CLM.GW_HDR_CASEID
    LEFT JOIN SourceStaging.VECCASRN.VEC_GW_TP_SUMMARY TPS ON TPS.CASEID = TP.ID
    LEFT JOIN SourceStaging.VECCASRN.VEC_GW_TPTYPE TPTYPE ON TPTYPE.ID = TPS.GW_TP_TYPEID
    WHERE UPPER(LTRIM(RTRIM(CLM.PRODUCT))) = 'MOTOR'
      AND NOT (TPTYPE.DISPLAY_STRING = 'Alleged Vandal'
               AND NOT EXISTS (SELECT 1 FROM dbo.LKP_EXPOSURE_CLAIM_EXCEPTIONS E WHERE E.CLAIM_REF = CLM.CLAIM_REF AND E.ExceptionType = 'VANDAL_CREATE_TP'))
      AND NOT (TPTYPE.DISPLAY_STRING = 'TP Driver (Thief)'
               AND NOT EXISTS (SELECT 1 FROM SourceStaging.VECCASRN.VEC_GW_PAY_TRANS PT WHERE PT.GW_HDR_CASEID = CLM.GW_HDR_CASEID)
               AND NOT EXISTS (SELECT 1 FROM dbo.LKP_EXPOSURE_CLAIM_EXCEPTIONS E WHERE E.CLAIM_REF = CLM.CLAIM_REF AND E.ExceptionType = 'THEFT_NO_TRANS_CREATE_TP'))
      AND NOT (TPTYPE.DISPLAY_STRING = 'TP Driver (Thief)'
               AND EXISTS (SELECT 1 FROM SourceStaging.VECCASRN.VEC_GW_PAY_TRANS PT WHERE PT.GW_HDR_CASEID = CLM.GW_HDR_CASEID)
               AND EXISTS (SELECT 1 FROM dbo.LKP_EXPOSURE_CLAIM_EXCEPTIONS E WHERE E.CLAIM_REF = CLM.CLAIM_REF AND E.ExceptionType = 'THEFT_WITH_TRANS_SUPPRESS_TP'))
),
TP_ELEMENTS AS (
    -- RESOLVED: the VEH/PRO combo question is settled per the mapping
    -- sheet's unconditional rule (no "still pending" marker found on
    -- either the sheet or CMP-86's own "Sign-Off" doc, so the sheet
    -- wins per your instruction). CMP-86's conditional Property
    -- restriction was never actually live in this code and has been
    -- removed rather than left as a confusing dead alternative.
    SELECT CLAIM_REF, ClaimPublicID, GW_HDR_CASEID, TP_ID, ClaimantRoleDesc, 'TP_VEH' AS RuleKey, 'VEH' AS ElementTag
    FROM TP_BASE
    WHERE VEHICLE = 'X'
       OR (COALESCE(VEHICLE,'') <> 'X' AND COALESCE(INJURY,'') <> 'X' AND COALESCE(PROPERTY,'') <> 'X')
    UNION ALL
    SELECT CLAIM_REF, ClaimPublicID, GW_HDR_CASEID, TP_ID, ClaimantRoleDesc, 'TP_INJ', 'INJ'
    FROM TP_BASE WHERE INJURY = 'X'
    UNION ALL
    SELECT CLAIM_REF, ClaimPublicID, GW_HDR_CASEID, TP_ID, ClaimantRoleDesc, 'TP_PRO', 'PRO'
    FROM TP_BASE WHERE PROPERTY = 'X'
)
INSERT INTO IntermediateStaging_DEV.dbo.IS_EXPOSURE_MOTOR
(PublicID, ClaimID, VectusCaseID_Adm, GW_HDR_CASEID, SourceOrigin_Adm, ExposureType, PrimaryCoverage,
 CoverageSubType, LossParty, ClaimOrder, CreateTime, CLAIM_REF, ClaimantType)
SELECT
    'mig:exp:tp' + LOWER(TE.ElementTag) + ':' + TE.ClaimPublicID + '_' + CONVERT(VARCHAR(64), TE.TP_ID),
    TE.ClaimPublicID, CONVERT(VARCHAR(64), TE.TP_ID), TE.GW_HDR_CASEID, TE.RuleKey,
    R.ExposureType, R.PrimaryCoverage, R.CoverageSubType, R.LossParty,
    0,
    (SELECT CONVERT(DATETIME2(7), CONCAT(VC.CREATE_DATE, ' ', VC.CREATE_TIME))
     FROM SourceStaging.VECCASRN.VEC_CASE VC WHERE VC.ID = TE.TP_ID),
    TE.CLAIM_REF,
    ISNULL(LK.CC_ClaimantType_Code, 'other')
FROM TP_ELEMENTS TE
INNER JOIN dbo.LKP_EXPOSURE_MOTOR_RULES R ON R.RuleKey = TE.RuleKey
LEFT JOIN dbo.LKP_TP_CLAIMANT_TYPE LK ON LK.VectusDisplayString = TE.ClaimantRoleDesc;


/* --------------------------------------------------------------
   BLOCK C: 3rd Party Hire & Mobility
   PAYMENT CODE LIST CONFLICT - see chat message.
   -------------------------------------------------------------- */
;WITH TP_HIRE_CANDIDATES AS (
    SELECT CLM.CLAIM_REF, CLM.PUBLICID AS ClaimPublicID, CLM.GW_HDR_CASEID, TP.ID AS TP_ID
    FROM IntermediateStaging_DEV.dbo.IS_CLAIM_MASTER CLM
    INNER JOIN SourceStaging.VECCASRN.VEC_GW_MOTOR_TP TP ON TP.GW_HDR_CASEID = CLM.GW_HDR_CASEID
    LEFT JOIN SourceStaging.VECCASRN.VEC_GW_TP_SUMMARY TPS ON TPS.CASEID = TP.ID
    WHERE UPPER(LTRIM(RTRIM(CLM.PRODUCT))) = 'MOTOR'
      AND (
            EXISTS (
                SELECT 1 FROM SourceStaging.VECCASRN.VEC_GW_PAY_TRANS PT
                INNER JOIN SourceStaging.VECCASRN.VEC_GW_PAY_DISS PD ON PD.GW_PAY_TRAN_ID = PT.ID
                WHERE PT.GW_HDR_CASEID = TP.GW_HDR_CASEID
                  AND PD.CODE IN ('ABH','COW','OUH','PLH','PLP','RVM','SUB','TEM','TPH','FSC','ABA','ABP')
            )
            OR (TPS.VEHICLE = 'X' AND EXISTS (SELECT 1 FROM SourceStaging.VECCASRN.VEC_GW_HIREREC HR WHERE HR.CASEID = TP.ID))
      )
)
INSERT INTO IntermediateStaging_DEV.dbo.IS_EXPOSURE_MOTOR

SELECT
    'mig:exp:tphire:' + H.ClaimPublicID + '_' + CONVERT(VARCHAR(64), H.TP_ID),

    ON LIAB.CASEID = CONVERT(BIGINT, E.VectusCaseID_Adm) AND LIAB.CURRENT_REC = 'X'
LEFT JOIN SourceStaging.dbo.TYPELIST_TABLE_MAPPING TL ON TL.Vectus_TypeCode = LIAB.LIAB_STATUS AND TL.TypeList_Name = 'LiabilityPosition_Adm'
WHERE E.SourceOrigin_Adm IN ('TP_VEH','TP_INJ','TP_PRO','TP_HIRE');

;WITH LATEST_RES_TRANS AS (
    SELECT RT.ID, RT.CASEID,
        ROW_NUMBER() OVER (PARTITION BY RT.CASEID ORDER BY RT.GORDER DESC) AS RN
    FROM SourceStaging.VECCASRN.VEC_GW_RES_TRANS RT
    WHERE RT.ACCEPTED = 'Y'
),
BI_RESERVE AS (
    SELECT E.PublicID, RA.CLAIM_LIFE, RA.LIABILITY, RA.YRS_TO_ISSUE
    FROM IntermediateStaging_DEV.dbo.IS_EXPOSURE_MOTOR E
    INNER JOIN LATEST_RES_TRANS LRT ON LRT.CASEID = CONVERT(BIGINT, E.VectusCaseID_Adm) AND LRT.RN = 1
    INNER JOIN SourceStaging.VECCASRN.VEC_GW_RES_ADJUST RA ON RA.GW_RES_TRANSID = LRT.ID
    WHERE E.SourceOrigin_Adm = 'TP_INJ'
)
UPDATE E
SET E.BIClaimLife_Adm = BR.CLAIM_LIFE,
    E.BIReservePercentage_Adm = BR.LIABILITY,
    E.BIYearsToIssue_Adm = BR.YRS_TO_ISSUE
FROM IntermediateStaging_DEV.dbo.IS_EXPOSURE_MOTOR E
INNER JOIN BI_RESERVE BR ON BR.PublicID = E.PublicID;


/* --------------------------------------------------------------
   BLOCK J: CILFigure_Adm / CILNotes_Adm
   -------------------------------------------------------------- */
;WITH CIL_TOTALS AS (
    SELECT PT.GW_HDR_CASEID, SUM(PD.AMOUNT) AS TotalCIL
    FROM SourceStaging.VECCASRN.VEC_GW_PAY_TRANS PT
    INNER JOIN SourceStaging.VECCASRN.VEC_GW_PAY_DISS PD ON PD.GW_PAY_TRAN_ID = PT.ID
    WHERE PD.CODE IN ('CIL','CIP')
    GROUP BY PT.GW_HDR_CASEID
)
UPDATE E
SET E.CILFigure_Adm = CT.TotalCIL, E.CILNotes_Adm = 'CIL'
FROM IntermediateStaging_DEV.dbo.IS_EXPOSURE_MOTOR E
INNER JOIN CIL_TOTALS CT ON CT.GW_HDR_CASEID = E.GW_HDR_CASEID;


/* --------------------------------------------------------------
   BLOCK K: State / CloseDate
   -------------------------------------------------------------- */
UPDATE E
SET E.State = TL.GW_TypeCode
FROM IntermediateStaging_DEV.dbo.IS_EXPOSURE_MOTOR E
INNER JOIN SourceStaging.VECCASRN.VEC_GW_CASE_STATUS CS
    ON CS.CASEID = CONVERT(BIGINT, E.VectusCaseID_Adm) AND CS.GCURRENT = 'X'
LEFT JOIN SourceStaging.dbo.TYPELIST_TABLE_MAPPING TL ON TL.Vectus_TypeCode = CS.STATUS AND TL.TypeList_Name = 'ExposureState'
WHERE E.SourceOrigin_Adm <> 'TP_HIRE' AND TL.GW_TypeCode IS NOT NULL;

;WITH HIRE_SETTLED AS (
    SELECT CASEID, MAX(ORDERING) AS MaxOrdering
    FROM SourceStaging.VECCASRN.VEC_HISTORY
    WHERE HISTORYTEXT = 'Hire claim settled'
    GROUP BY CASEID
),
HIRE_SETTLED_DETAIL AS (
    SELECT H.CASEID, H.CREATEDATE, H.CREATETIME
    FROM SourceStaging.VECCASRN.VEC_HISTORY H
    INNER JOIN HIRE_SETTLED HS ON HS.CASEID = H.CASEID AND HS.MaxOrdering = H.ORDERING
),
HIRE_CASE_STATUS AS (
    SELECT CS.CASEID, CS.STATUS
    FROM SourceStaging.VECCASRN.VEC_GW_CASE_STATUS CS
    WHERE CS.GCURRENT = 'X'
)
UPDATE E
SET E.State = CASE
        WHEN HCS.STATUS = 'Finalised' THEN 'closed'
        WHEN HSD.CASEID IS NOT NULL THEN 'closed'
        ELSE 'open'
     END,
    E.CloseDate = CASE
        WHEN HSD.CASEID IS NOT NULL THEN CONVERT(DATETIME2(7), CONCAT(HSD.CREATEDATE, ' ', HSD.CREATETIME))
        ELSE NULL
     END
FROM IntermediateStaging_DEV.dbo.IS_EXPOSURE_MOTOR E
LEFT JOIN HIRE_CASE_STATUS HCS ON HCS.CASEID = CONVERT(BIGINT, E.VectusCaseID_Adm)
LEFT JOIN HIRE_SETTLED_DETAIL HSD ON HSD.CASEID = CONVERT(BIGINT, E.VectusCaseID_Adm)
WHERE E.SourceOrigin_Adm = 'TP_HIRE';

UPDATE E
SET E.CloseDate = CS.RECORD_DATE
FROM IntermediateStaging_DEV.dbo.IS_EXPOSURE_MOTOR E
INNER JOIN SourceStaging.VECCASRN.VEC_GW_CASE_STATUS CS
    ON CS.CASEID = CONVERT(BIGINT, E.VectusCaseID_Adm) AND CS.GCURRENT = 'X' AND CS.STATUS = 'Finalised'
WHERE E.SourceOrigin_Adm <> 'TP_HIRE' AND E.State = 'closed';

END

   OR ClaimOrder IS NULL OR AssignmentStatus IS NULL OR BIReservePercentage_Adm IS NULL
   OR RIGroupSetExternally IS NULL OR State IS NULL OR SupplementalWorkloadWeight IS NULL
   OR WorkloadWeight IS NULL OR CreateTime IS NULL;

FROM SourceStaging.VECCASRN.VEC_GW_MOTOR_AD

HAVING COUNT(DISTINCT ID) > 1;

WHERE CLAIM_REF IS NOT NULL AND Solution LIKE 'Migrate-group%'
AND CLAIM_REF NOT IN (SELECT CLAIM_REF FROM IntermediateStaging_DEV.dbo.IS_EXPOSURE_MOTOR WHERE SourceOrigin_Adm = 'AD');

FROM IntermediateStaging_DEV.dbo.IS_EXPOSURE_MOTOR
GROUP BY SourceOrigin_Adm;

FROM IntermediateStaging_DEV.dbo.IS_CLAIM_MASTER CLM
INNER JOIN SourceStaging.VECCASRN.VEC_GW_MOTOR_TP TP ON TP.GW_HDR_CASEID = CLM.GW_HDR_CASEID
WHERE UPPER(LTRIM(RTRIM(CLM.PRODUCT))) = 'MOTOR'
GROUP BY CLM.CLAIM_REF
HAVING COUNT(*) > 1;

FROM SourceStaging.VECCASRN.VEC_GW_TP_SUMMARY S
INNER JOIN SourceStaging.VECCASRN.VEC_GW_TPTYPE T ON T.ID = S.GW_TP_TYPEID
WHERE NOT EXISTS (SELECT 1 FROM dbo.LKP_TP_CLAIMANT_TYPE LK WHERE LK.VectusDisplayString = T.DISPLAY_STRING);
