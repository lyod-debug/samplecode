       LTRIM(RTRIM(C.CLAIM_REF)) AS Claim_Number,
       UPPER(LTRIM(RTRIM(C.PRODUCT))) AS LOBCode,
       'Date of Loss' AS Parameter_Name,
       CAST(CHAR(C.ACTUAL_DOL, ISO) AS VARCHAR(255)) AS V2_value   -- CONFIRM CHAR() syntax against DB2 (see Part 4.5)

WHERE UPPER(TRIM(RTRIM(C.GCURRENT))) = 'X'
  AND (C.INCIDENT_CODE IS NULL OR UPPER(TRIM(RTRIM(C.INCIDENT_CODE))) <> 'WINDSCREEN')
  AND NULLIF(TRIM(RTRIM(C.CLAIM_REF)), '') IS NOT NULL

-- #5: Claim Finalised Date. WHERE fixed per Part 4.4 (CONFIRM this fix is correct):
SELECT (EXECID), LTRIM(RTRIM(C.CLAIM_REF)), UPPER(LTRIM(RTRIM(C.PRODUCT))),
       'Claim Finalised date',
       CAST(CASE WHEN CS.RECORD_DATE IS NOT NULL AND CS.RECORD_TIME IS NOT NULL
                 THEN VARCHAR_FORMAT(TIMESTAMP(CS.RECORD_DATE, CS.RECORD_TIME), 'YYYY-MM-DD HH24:MI:SS')
            END AS VARCHAR(255))
FROM (
    SELECT C.*, CS.RECORD_DATE, CS.RECORD_TIME,
           ROW_NUMBER() OVER (PARTITION BY LTRIM(RTRIM(C.CLAIM_REF))
                               ORDER BY CS.RECORD_DATE DESC, CS.RECORD_TIME DESC) AS rn
    FROM (SCHEMA).VEC_GW_CLAIM_SUM C
    INNER JOIN (SCHEMA).VEC_GW_CLAIM_STATUS CS ON CS.CASEID = C.GW_HDR_CASEID
    WHERE UPPER(TRIM(RTRIM(C.GCURRENT))) = 'X'
      AND (C.INCIDENT_CODE IS NULL OR UPPER(TRIM(RTRIM(C.INCIDENT_CODE))) <> 'WINDSCREEN')
      AND NULLIF(TRIM(RTRIM(C.CLAIM_REF)), '') IS NOT NULL
      AND UPPER(TRIM(RTRIM(CS.GCURRENT))) = 'X'                 -- CONFIRM: inferred fix, see Part 4.4
      AND UPPER(TRIM(RTRIM(CS.STATUS))) = 'FINALISED'
) C WHERE rn = 1

UNION ALL
