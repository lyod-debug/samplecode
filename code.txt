CREATE TABLE audit.CCST_VolumeRecon_Landing
(
    Execution_id     BIGINT       NOT NULL,
    Recon_id         INT          NOT NULL,
    Parameter_Value  VARCHAR(100) NULL,     -- e.g. 'HOUSEHOLD', 'MOTOR' — must match how ClaimVolumeReconciliationData.Parameter_value is stored
    CCST_Value       INT          NULL,
    Created_at       DATETIME     NOT NULL DEFAULT GETDATE()
);


SELECT (EXECID) AS Execution_id,
       106 AS Recon_id,
       CASE
           WHEN P.PolicyType = 'PHPERSONALHOME' THEN 'HOUSEHOLD'
           WHEN P.PolicyType = 'PMPERSONALMOTOR' THEN 'MOTOR'
       END AS Parameter_Value,
       COUNT(DISTINCT C.PublicID) AS CCST_Value
FROM gwccdv24.dbo.ccst_claim C
INNER JOIN gwccdv24.dbo.ccst_policy P ON P.PublicID = C.PolicyID
WHERE C.PublicID IS NOT NULL
  AND P.PolicyType IS NOT NULL
GROUP BY
    CASE
        WHEN P.PolicyType = 'PHPERSONALHOME' THEN 'HOUSEHOLD'
        WHEN P.PolicyType = 'PMPERSONALMOTOR' THEN 'MOTOR'
    END

-- Repeat for every other Recon_id block (101–105, etc.) that exists in
-- ClaimVolumeReconciliationData, using the corresponding gwccdv24 tables/filters
-- that mirror each block's IS-side logic.


CREATE OR ALTER PROCEDURE audit.usp_VolumeRecon_CCST
    @Execution_id BIGINT
AS
BEGIN
    SET NOCOUNT ON;

    UPDATE R
    SET
        R.CCST_value = X.CCST_Value,
        R.IS_CCST_result =
            CASE
                WHEN R.IS_value IS NULL THEN NULL
                WHEN R.IS_value = X.CCST_Value THEN 'PASS'
                ELSE 'FAIL'
            END
    FROM audit.CCST_VolumeRecon_Landing X
    INNER JOIN [audit].ClaimVolumeReconciliationData R
        ON X.Parameter_Value = UPPER(LTRIM(RTRIM(R.Parameter_value)))
       AND X.Recon_id = R.Recon_id
    WHERE R.Execution_id = @Execution_id
      AND X.Execution_id = @Execution_id
      AND X.Recon_id = 106;

    -- Repeat this block (or parameterize by @Recon_id) for every other volume recon check
END
