/*=========================================================
  207 - LossCause : HOUSEHOLD
=========================================================*/
Insert into [audit].[Tmp_Attr_Recon]
(Matrixname, recon_ID, LOB, SS_IS_Match, SS_IS_MisMatch) values
(
    'LossCause'
    ,207
    ,'HOUSEHOLD'

    ,( Select count(*) from (
        Select C.CLAIM_REF, CAUSE_CIRC.[cctl_losscause_TYPECODE] AS LossCause
        FROM [SourceStaging].VECCASRN.VEC_GW_CLAIM_SUM C
        join [SourceStaging].VECCASRN.VEC_GWP_POLICY P on C.GW_HDR_CASEID = P.GW_HDR_CASEID
        LEFT JOIN [SourceStaging].VECCASRN.VEC_HH_INC_DETS D ON D.HH_CLAIMID = C.GW_HDR_CASEID
        LEFT JOIN [SourceStaging].VECCASRN.VEC_HH_AXX_CIRCS CIRC_LKP  ON D.CIRC  = CIRC_LKP.ID
        LEFT JOIN [SourceStaging].VECCASRN.VEC_HH_AXX_CIRCS CAUSE_LKP ON D.CAUSE = CAUSE_LKP.ID
        LEFT JOIN [SourceStaging].[dbo].[CLAIM_LOSSCAUSE_CIRC_MAPPING] CAUSE_CIRC
            ON CAUSE_LKP.CAUSE = CAUSE_CIRC.Vectus_CAUSE
           AND CIRC_LKP.CIRC  = CAUSE_CIRC.Vectus_CIRCUMSTANCE
        where LTRIM(RTRIM(C.GCURRENT)) = 'X'
          and p.PRODUCT = 'HOUSEHOLD'

        INTERSECT

        SELECT c.ClaimNumber, c.LossCause      -- CONFIRM real column name on dbo.IS_CLAIM
            FROM dbo.IS_CLAIM c
                JOIN dbo.IS_POLICY p on c.PolicyID = p.PUBLICID
            WHERE c.PublicID IS NOT NULL
              and POLICYTYPE = 'PHPersonalHome'
        ) A
    )

    ,( Select count(*) from (
        Select C.CLAIM_REF, CAUSE_CIRC.[cctl_losscause_TYPECODE] AS LossCause
        FROM [SourceStaging].VECCASRN.VEC_GW_CLAIM_SUM C
        join [SourceStaging].VECCASRN.VEC_GWP_POLICY P on C.GW_HDR_CASEID = P.GW_HDR_CASEID
        LEFT JOIN [SourceStaging].VECCASRN.VEC_HH_INC_DETS D ON D.HH_CLAIMID = C.GW_HDR_CASEID
        LEFT JOIN [SourceStaging].VECCASRN.VEC_HH_AXX_CIRCS CIRC_LKP  ON D.CIRC  = CIRC_LKP.ID
        LEFT JOIN [SourceStaging].VECCASRN.VEC_HH_AXX_CIRCS CAUSE_LKP ON D.CAUSE = CAUSE_LKP.ID
        LEFT JOIN [SourceStaging].[dbo].[CLAIM_LOSSCAUSE_CIRC_MAPPING] CAUSE_CIRC
            ON CAUSE_LKP.CAUSE = CAUSE_CIRC.Vectus_CAUSE
           AND CIRC_LKP.CIRC  = CAUSE_CIRC.Vectus_CIRCUMSTANCE
        where LTRIM(RTRIM(C.GCURRENT)) = 'X'
          and p.PRODUCT = 'HOUSEHOLD'

        EXCEPT

        SELECT c.ClaimNumber, c.LossCause
            FROM dbo.IS_CLAIM c
                JOIN dbo.IS_POLICY p on c.PolicyID = p.PUBLICID
            WHERE c.PublicID IS NOT NULL
              and POLICYTYPE = 'PHPersonalHome'
        ) B
    )
),

              and POLICYTYPE = 'PMPersonalMotor'

    ,( Select count(*) from (

        LEFT JOIN [SourceStaging].VECCASRN.VEC_GW_CIRCS_REF REF ON REF.ID = INC.GW_CIRCS_REFID

            ON REF.LOSSCAUSE     = MAP.VECTUS_LOSSCAUSE
           AND REF.CIRCUMSTANCE  = MAP.VECTUS_CIRCUMSTANCE
        where LTRIM(RTRIM(C.GCURRENT)) = 'X'
          and p.PRODUCT = 'MOTOR'

        EXCEPT

        SELECT c.ClaimNumber, c.LossCause
            FROM dbo.IS_CLAIM c
                JOIN dbo.IS_POLICY p on c.PolicyID = p.PUBLICID
            WHERE c.PublicID IS NOT NULL
              and POLICYTYPE = 'PMPersonalMotor'
        ) B
