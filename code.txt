/* Update NonFinReconciliationresult Table */

UPDATE Res
SET Res.SS_Transformed_Count = X.SSTransformedCnt,
    Res.IS_Count             = X.ISCnt,
    Res.SS_IS_Match          = X.MatchCnt,
    Res.SS_IS_Mismatch       = X.MismatchCnt,
    Res.SS_IS_result         = CASE WHEN X.MismatchCnt = 0 THEN 'PASS' ELSE 'FAIL' END

FROM IntermediateStaging_DEV.[audit].NonFinReconciliationresult AS Res
INNER JOIN
(
    SELECT Recon_id, Parameter_Name, LOBCode,

        COUNT(1) AS SSTransformedCnt,
        COUNT(1) AS ISCnt,
        SUM(CASE WHEN SS_IS_result = 'PASS' THEN 1 ELSE 0 END) AS MatchCnt,
        SUM(CASE WHEN SS_IS_result = 'FAIL' THEN 1 ELSE 0 END) AS MismatchCnt

    FROM #Compared2
    GROUP BY Recon_id, Parameter_Name, LOBCode

) AS X
ON Res.Execution_id  = @Execution_id
AND Res.Recon_id      = X.Recon_id
AND Res.Parameter_Name = X.Parameter_Name
AND Res.LOBCode        = X.LOBCode;
